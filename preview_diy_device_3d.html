───────────────────────────────────────────────────────────────────
1. Màn hình OLED SSD1306 (Giao tiếp I2C):
OLED GND ───► ESP32 GND
OLED VCC ───► ESP32 3V3 (3.3V)
OLED SCL ───► ESP32 GPIO 22 (I2C Clock)
OLED SDA ───► ESP32 GPIO 21 (I2C Data)
───────────────────────────────────────────────────────────────────
2. Bộ 3 Phím Bấm Điều Hướng (Cấu hình INPUT_PULLUP):
Nút 1 [◀ Prev] ───► ESP32 GPIO 18 (Chân còn lại nối GND)
Nút 2 [👁 Flip] ───► ESP32 GPIO 19 (Chân còn lại nối GND)
Nút 3 [Next ▶] ───► ESP32 GPIO 23 (Chân còn lại nối GND)
───────────────────────────────────────────────────────────────────
3. Mạch Nguồn & Pin:
Cực (+) Pin Li-Po ───► Công tắc gạt ───► Chân VIN / 5V của ESP32
Cực (-) Pin Li-Po ───► Chân GND của ESP32

Mã nguồn C++ Arduino IDE hoàn chỉnh (sử dụng thư viện Adafruit SSD1306) để nạp vào ESP32 ngoài đời thực:

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

const int BTN_PREV = 18;
const int BTN_FLIP = 19;
const int BTN_NEXT = 23;

struct WordItem {
  const char* word;
  const char* ipa;
  const char* type;
  const char* meaning;
};

WordItem vocabList[] = {
  {"APPROVE", "/ə'pru:v/", "v.", "Phe duyet, chap thuan"},
  {"INVOICE", "/'invɔis/", "n.", "Hoa don thanh toan"},
  {"NEGOTIATE", "/ni'gəʊʃieit/", "v.", "Dam phan, thuong luong"},
  {"RELIABLE", "/ri'laiəbl/", "adj.", "Dang tin cay"}
};
int currentIndex = 0;
bool isFlipped = false;

void setup() {
  pinMode(BTN_PREV, INPUT_PULLUP);
  pinMode(BTN_FLIP, INPUT_PULLUP);
  pinMode(BTN_NEXT, INPUT_PULLUP);

  Wire.begin(21, 22);
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  renderScreen();
}

void renderScreen() {
  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.print("VOCAB POCKET  BAT:98%");
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);

  if (!isFlipped) {
    display.setTextSize(2);
    display.setCursor(4, 20);
    display.print(vocabList[currentIndex].word);
    display.setTextSize(1);
    display.setCursor(4, 42);
    display.print(vocabList[currentIndex].ipa);
  } else {
    display.setTextSize(1);
    display.setCursor(4, 18);
    display.print(vocabList[currentIndex].type);
    display.setTextSize(1);
    display.setCursor(4, 30);
    display.print(vocabList[currentIndex].meaning);
  }

  display.drawLine(0, 52, 128, 52, SSD1306_WHITE);
  display.setCursor(0, 55);
  display.print("[<]Prev  [*]Flip  [>]Next");
  display.display();
}

void loop() {
  if (digitalRead(BTN_NEXT) == LOW) {
    currentIndex = (currentIndex + 1) % 4;
    isFlipped = false;
    renderScreen();
    delay(200);
  }
  if (digitalRead(BTN_PREV) == LOW) {
    currentIndex = (currentIndex - 1 + 4) % 4;
    isFlipped = false;
    renderScreen();
    delay(200);
  }
  if (digitalRead(BTN_FLIP) == LOW) {
    isFlipped = !isFlipped;
    renderScreen();
    delay(200);
  }
}
